3b) grammer with symbol table 

input : She and he walk with a smart man in the city wow

code : 

%{
#include <stdio.h>
#include <string.h>
#include <ctype.h>

char symtab[200][50];
int count = 0;

void toLower(char *word) {
    for(int i=0; word[i]; i++)
        word[i] = tolower(word[i]);
}

void add(char *word) {
    char temp[50];
    strcpy(temp, word);
    toLower(temp);

    for(int i=0;i<count;i++) {
        if(strcmp(symtab[i], temp)==0)
            return;
    }
    strcpy(symtab[count++], temp);
}
%}

%%
